Schema

CREATE TABLE favicons (
  id          INTEGER PRIMARY KEY,
  host        TEXT    NOT NULL UNIQUE,    -- www-stripped
  data        BLOB    NOT NULL,
  data_type   TEXT    NOT NULL,           -- png | svg | ico | webp | jpeg | gif
  width       INTEGER NOT NULL DEFAULT 0, -- best quality wins (svg = 0)
  updated_at  INTEGER NOT NULL            -- epoch ms · harvest time
);
